/**
 * The maze grid around one shaft, seen as a ring of 48 blocks.
 * Block 0 is the corner at offset (-6, 6). The ring runs along the face where the second offset is 6.
 * Nodes sit on even blocks, so node {@code v} is block {@code 2v}.
 *
 * <p>Written by Claude Code; conversation transcript: {@code llm/transcripts/transcript_2026-10-29_07-05_5f965fb4-ad37-4b6a-8b5d-da8c3a179dc6.md}.
 */
public final class Ring {

    public static final int BLOCKS = 48;
    public static final int NODES = BLOCKS / 2;
    public static final int FACE_NODES = 6;
    /** The offset of the outer layer from the shaft center. */
    public static final int OUTER = 6;
    /** The offset of the junction box face from the junction center. */
    public static final int BOX = 7;

    private Ring() {
    }

    public static boolean isCorner(int node) {
        return node % FACE_NODES == 0;
    }

    public static int first(int block) {
        block = Math.floorMod(block, BLOCKS);
        if (block < 12) {
            return -OUTER + block;
        }
        if (block < 24) {
            return OUTER;
        }
        if (block < 36) {
            return OUTER - (block - 24);
        }
        return -OUTER;
    }

    public static int second(int block) {
        block = Math.floorMod(block, BLOCKS);
        if (block < 12) {
            return OUTER;
        }
        if (block < 24) {
            return OUTER - (block - 12);
        }
        if (block < 36) {
            return -OUTER;
        }
        return -OUTER + (block - 36);
    }

    /** The ring block at the offset, or -1 when the offset is not on the outer layer. */
    public static int block(int first, int second) {
        if (second == OUTER && first >= -OUTER && first < OUTER) {
            return first + OUTER;
        }
        if (first == OUTER && second > -OUTER && second <= OUTER) {
            return 12 + (OUTER - second);
        }
        if (second == -OUTER && first > -OUTER && first <= OUTER) {
            return 24 + (OUTER - first);
        }
        if (first == -OUTER && second >= -OUTER && second < OUTER) {
            return 36 + (second + OUTER);
        }
        return -1;
    }

    /** The node on the face where the second offset is {@code 6 * sign}, at the first offset {@code 2k}. */
    public static int faceNode(int sign, int k) {
        return block(2 * k, OUTER * sign) / 2;
    }

    /** The ring block between two neighbor nodes. */
    public static int blockBetween(int node1, int node2) {
        return Math.abs(node1 - node2) == 1 ? 2 * Math.min(node1, node2) + 1 : BLOCKS - 1;
    }

    public static int oddBelow(int a) {
        return (a & 1) != 0 ? a : a - 1;
    }

    public static int oddAbove(int a) {
        return (a & 1) != 0 ? a : a + 1;
    }

}

 * An undirected graph whose adjacency lists keep the insertion sequence, with the Origin Shift sampler.
 *
 * <p>Written by Claude Code; conversation transcript: {@code llm/transcripts/transcript_2026-10-29_07-05_5f965fb4-ad37-4b6a-8b5d-da8c3a179dc6.md}.

public final class MazeGraph {

    /** The Origin Shift steps per node. The stats mode of the maze prototype matches Wilson's algorithm at this count. */
    public static final int STEPS_PER_NODE = 16;

    private final List<IntList> adjacency = new ArrayList<>();

    public int addNode() {
        this.adjacency.add(new IntArrayList(4));
        return this.adjacency.size() - 1;
    }

    public void addEdge(int a, int b) {
        this.adjacency.get(a).add(b);
        this.adjacency.get(b).add(a);
    }

    public int size() {
        return this.adjacency.size();
    }

    /**
     * Samples a spanning tree with Origin Shift from the breadth-first tree of a random root.
     *
     * @return the parent of each node, -1 for the root
     */
    public int[] sampleTree(RandomSource random) {
        int root = random.nextInt(this.size());
        int[] parent = this.breadthFirstTree(root);
        long steps = (long) STEPS_PER_NODE * this.size();
        for (long step = 0; step < steps; step++) {
            IntList neighbors = this.adjacency.get(root);
            int next = neighbors.getInt(random.nextInt(neighbors.size()));
            parent[root] = next;
            root = next;
            parent[root] = -1;
        }
        return parent;
    }

    private int[] breadthFirstTree(int root) {
        int[] parent = new int[this.size()];
        java.util.Arrays.fill(parent, -2);
        parent[root] = -1;
        int[] queue = new int[this.size()];
        int head = 0;
        int tail = 0;
        queue[tail++] = root;
        while (head < tail) {
            int node = queue[head++];
            IntList neighbors = this.adjacency.get(node);
            for (int k = 0; k < neighbors.size(); k++) {
                int next = neighbors.getInt(k);
                if (parent[next] == -2) {
                    parent[next] = node;
                    queue[tail++] = next;
                }
            }
        }
        return parent;
    }

}

    /** Collects the tree edges of a sampled graph. A node without a name is a virtual node. */
    static List<MazeEdge> treeEdges(int[] parent, IntFunction<MazeNode> names, IntFunction<MazeNode> virtualTargets) {
        List<MazeEdge> edges = new ArrayList<>();
        for (int node = 0; node < parent.length; node++) {
            int p = parent[node];
            if (p < 0) {
                continue;
            }
            MazeNode a = names.apply(node);
            MazeNode b = names.apply(p);
            if (a == null) {
                a = virtualTargets.apply(node);
            }
            if (b == null) {
                b = virtualTargets.apply(p);
            }
            edges.add(new MazeEdge(a, b));
        }
        return edges;
    }

    /** Drops every corner node of degree one together with its edge. */
    static List<MazeEdge> pruneCornerLeaves(List<MazeEdge> edges) {
        Map<MazeNode, Integer> degree = new HashMap<>();
        for (MazeEdge edge : edges) {
            degree.merge(edge.a, 1, Integer::sum);
            degree.merge(edge.b, 1, Integer::sum);
        }
        List<MazeEdge> kept = new ArrayList<>(edges.size());
        for (MazeEdge edge : edges) {
            if (!isCornerLeaf(edge.a, degree) && !isCornerLeaf(edge.b, degree)) {
                kept.add(edge);
            }
        }
        return kept;
    }

    private static boolean isCornerLeaf(MazeNode node, Map<MazeNode, Integer> degree) {
        return node instanceof MazeNode.Ruling ruling && ruling.isCorner() && degree.get(node) == 1;
    }

final class RulingGraph {

    final MazeGraph graph = new MazeGraph();
    private final List<MazeNode> names = new ArrayList<>();
    private final Int2IntMap firstNodeOfRuling = new Int2IntOpenHashMap();

    RulingGraph(int[] rulings) {
        for (int along : rulings) {
            this.firstNodeOfRuling.put(along, this.graph.size());
            for (int v = 0; v < Ring.NODES; v++) {
                this.add(new MazeNode.Ruling(along, v));

     */
    public FractalNoise(RandomSource random, int octaveCount, double scale) {
        this.octaves = new SimplexNoise[octaveCount];
        double sum = 0;
        for (int k = 0; k < octaveCount; k++) {
            this.octaves[k] = new SimplexNoise(random);
            sum += 1.0 / (1 << k);

        this.frequency = 1.0 / scale;

        double amplitude = 1;

            value += amplitude * octave.getValue(x * frequency, y * frequency, z * frequency);
            frequency *= 2;
            amplitude *= 0.5;
        }
        return value / this.amplitudeSum;

final class MazeCarver {

    /**
     * One maze path block.
     *
     * @param surface the maze grid block, which becomes air
     * @param beneath the block under it, which becomes the maze path block
     * @param normal  the outward face direction when the block is on a patch

    record Carve(BlockPos surface, BlockPos beneath, @Nullable Direction normal) {

    /** A patch face: the frame of its horizontal shaft, the face sign and the junction center along that shaft. */
    record PatchFace(ShaftFrame frame, int sign, int center) {

        Carve carve(int along, int first) {
            return new Carve(
                    this.frame.at(along, first, Ring.BOX * this.sign),
                    this.frame.at(along, first, (Ring.BOX - 1) * this.sign),
                    this.frame.secondDirection(this.sign)
            );
        }

        Carve node(int i, int j) {
            return this.carve(this.center + 2 * i, 2 * j);

    }

    static Carve block(ShaftFrame frame, int along, int block) {
        int first = Ring.first(block);
        int second = Ring.second(block);
        return new Carve(frame.at(along, first, second), frame.at(along, clampInterior(first), clampInterior(second)), null);
    }

    private static int clampInterior(int offset) {
        return Math.max(-Ring.OUTER + 1, Math.min(Ring.OUTER - 1, offset));
    }

    /** Carves the ruling nodes of the rulings. A corner node carves only when an edge holds it. */
    static void rulings(ShaftFrame frame, int[] rulings, List<MazeEdge> edges, Consumer<Carve> out) {
        Set<MazeNode> present = new HashSet<>();
        for (MazeEdge edge : edges) {
            present.add(edge.a());
            present.add(edge.b());
        }
        for (int along : rulings) {
            for (int v = 0; v < Ring.NODES; v++) {
                if (!Ring.isCorner(v) || present.contains(new MazeNode.Ruling(along, v))) {
                    out.accept(block(frame, along, 2 * v));
                }
            }
        }

        if (a.along() == b.along()) {
            out.accept(block(frame, a.along(), Ring.blockBetween(a.node(), b.node())));
        } else {
            out.accept(block(frame, Math.min(a.along(), b.along()) + 1, 2 * a.node()));

/**
 * Confirms that a set of maze edges forms one tree.
 *
 * <p>Written by Claude Code; conversation transcript: {@code llm/transcripts/transcript_2026-10-29_07-05_5f965fb4-ad37-4b6a-8b5d-da8c3a179dc6.md}.
 */
final class TreeCheck {

    private final Map<MazeNode, MazeNode> parent = new HashMap<>();

    private TreeCheck() {
    }

    private MazeNode find(MazeNode node) {
        MazeNode root = node;
        while (true) {
            MazeNode next = this.parent.getOrDefault(root, root);
            if (next.equals(root)) {
                break;
            }
            root = next;
        }
        this.parent.put(node, root);
        return root;
    }

    /** Asserts no loop and one component. Returns the node count. */
    static int assertOneTree(Collection<MazeEdge> edges) {
        TreeCheck check = new TreeCheck();
        Set<MazeNode> nodes = new HashSet<>();
        for (MazeEdge edge : edges) {
            nodes.add(edge.a());
            nodes.add(edge.b());
            MazeNode a = check.find(edge.a());
            MazeNode b = check.find(edge.b());
            assertNotEquals(a, b, () -> "The edge " + edge + " closes a loop");
            check.parent.put(a, b);
        }
        Set<MazeNode> roots = new HashSet<>();
        for (MazeNode node : nodes) {
            roots.add(check.find(node));
        }
        assertEquals(1, roots.size(), "The edges form more than one component");
        assertEquals(nodes.size() - 1, edges.size());
        return nodes.size();
    }

}

### Claude:

I added the Aerolite world preset and the shafts chunk step, and committed them on `claude/world-preset`. The customize screen with the two decay sliders compiles and the client reaches the main menu with it. I have not clicked through that screen, because I can't drive the client GUI from here.

**What it does**
- **Preset:** `aerolite:aerolite` is in `#minecraft:normal`. The Overworld, the Nether and the End each get `AeroliteChunkGenerator`, a subclass of the noise-based chunk generator. The generator carries the settings.
- **Terrain:** the noise settings make only air, as you chose.
- **Chunk step:** two mixins put `aerolite:shafts` between carvers and features in both chunk pyramids. Features wait for the shafts of their neighbor chunks, so ores and other features land inside the shafts. No API adds a chunk status, so mixins are the only way.
- **Shafts:**
  - Placement uses a jittered grid, with at least 16 blocks between two shafts on the same axis.
  - A crossing gets no column when no ruling fits between its two junction boxes.
  - Each column end gets a random shear band of 8 to 64 blocks.
  - The full maze is in: segments, junction segments, column pieces and patches.
- **Decay:** the threshold t is a setting, default 0.5. On a patch, the box block that decay exposes becomes cobblestone.
- **Sliders:** the customize screen has a decay scale slider (powers of two from 32 to 1024) and a threshold slider (0 to 1 in steps of 0.05).

**Choices you did not make**
- **Patch decay reading:** I took "place cobblestone instead of junction block" to mean the exposed box block becomes the path block. I also kept the one-block rim of each patch face as wall, so decay does not touch it.
- **Merged junction segments:** with only a 16-block gap, the junction segments of two columns on one shaft can share a chunk. Then one junction segment spans both boxes. I added this case to the prototype first (`junction.py --cluster-stats`). All 600 random scenes stayed one tree.
- **Overworld sea level:** the first test world had a lava floor at the bottom. Vanilla fills air below Y -54 with lava when the sea level is higher than that. I set the void Overworld's sea level to -64, the same value vanilla's floating islands preset uses.
- **Small constants:** decay is 4 octaves of simplex noise. Shear noise is 3 octaves at a 16-block scale.

These decisions are now in CLAUDE.md.

**Test results**
- **Build:** compile and the full build pass, and data generation writes the expected files. No test or data generation classes are in the production jar.
- **JUnit:** 20 tests pass. They cover one tree per shaft, column and merged cluster, the 16-block gap, the slots and the settings codec.
- **GameTests:** 3 pass. They check the step order, the blocks the painter writes, and that the preset loads from the data pack and builds shafts.
- **Real world:** a dev-server world with the preset generated its spawn area in about 2 seconds. 58 saved chunks hold shafts at the `aerolite:shafts` status. That run left a `server.properties` set to the preset in the worktree's `run/` folder, which is gitignored.
- **Renders:** I rendered junctions from the real painter. They match the prototype images, and decay shows on the shafts and the patches.
